# dedupe itself declares no upper bounds on its own dependencies (only
# lower bounds / none at all), so pip will happily install versions of its
# transitive deps that are NOT actually compatible with dedupe==3.0.3's
# internal code. These pins are required, not optional, for a fresh
# `pip install -r requirements.txt` to produce a WORKING dedupe install.
dedupe==3.0.3
numpy==2.4.4
scikit-learn==1.8.0

# THE ROOT-CAUSE PIN: BTrees 6.5 (2026-08-20) removed the byValue() method
# from every Bucket/BTree family except II/LL (see BTrees CHANGES.rst,
# "Remove the long deprecated byValue() method..."). dedupe's own
# canopy_index.py calls results.byValue(...) on an IFBucket (int keys,
# float values) -- a family NOT in the II/LL exemption -- so ANY BTrees
# >=6.5 breaks dedupe training with:
#   AttributeError: 'BTrees.IFBTree.IFBucket' object has no attribute 'byValue'
# BTrees 6.4 is the newest release that still has byValue() on IFBucket.
BTrees==6.4
